    52→export interface CommitGraphProps {
    53→  commits: GraphCommit[]
    54→  loading: boolean
    55→  fetching?: boolean
    56→  error: Error | null
    57→  onRetry?: () => void
    58→  selectedHash: string | null
    59→  onSelect: (hash: string | null) => void
    60→  currentBranch: string
    61→  /** toolbar slot: branch filter, search, … */
    62→  extra?: ReactNode
    63→}
    64→
    65→function clamp(v: number, min: number, max: number): number {
    66→  return Math.min(max, Math.max(min, v))
    67→}
    68→
    69→/** list-column time: HH:mm for recent commits, date for older ones */
    70→function rowTime(iso: string): string {
    71→  const d = new Date(iso)
    72→  return Date.now() - d.getTime() < 86_400_000
    73→    ? format(d, 'HH:mm')
    74→    : format(d, 'yyyy-MM-dd')
    75→}

    78→/*  Commit Graph — the interactive heart of the application            */
    79→/* ------------------------------------------------------------------ */
    80→
    81→export function CommitGraph({
    82→  commits,
    83→  loading,
    84→  fetching,
    85→  error,
    86→  onRetry,
    87→  selectedHash,
    88→  onSelect,
    89→  currentBranch,
    90→  extra,
    91→}: CommitGraphProps) {
    92→  const containerRef = useRef<HTMLDivElement>(null)
    93→  const tooltipRef = useRef<HTMLDivElement>(null)
    94→
    95→  const [size, setSize] = useState({ w: 0, h: 0 })
    96→  const [t, setT] = useState<Transform>({ scale: 1, tx: 0, ty: 0 })
    97→  const [hoveredRow, setHoveredRow] = useState<number | null>(null)
    98→  const [intro, setIntro] = useState(false)
    99→
   100→  const tRef = useRef(t)
   101→  const sizeRef = useRef(size)
   102→  const commitsRef = useRef(commits)
   103→
   104→  const layout: GraphLayout | null = useMemo(
   105→    () => (commits.length > 0 ? computeGraphLayout(commits) : null),
   106→    [commits],
   107→  )
   108→  const layoutRef = useRef(layout)
   109→
   110→  /* ---------------- drag / pinch state (refs, no re-render) --------- */
   111→  const pointers = useRef(new Map<number, { x: number; y: number }>())
   112→  const panStart = useRef<{ tx: number; ty: number; x: number; y: number } | null>(null)
   113→  const pinchStart = useRef<{
   114→    dist: number
   115→    scale: number
   116→    midX: number
   117→    midY: number
   118→    tx: number
   119→    ty: number
   120→  } | null>(null)
   121→  const didPan = useRef(false)
   122→  const panAnim = useRef<number | null>(null)
   123→  const moveScheduled = useRef(false)
   124→
   125→  const cancelPanAnim = useCallback(() => {
   126→    if (panAnim.current !== null) {
   127→      cancelAnimationFrame(panAnim.current)
   128→      panAnim.current = null
   129→    }
   130→  }, [])
   131→
   132→  /* ---------------- geometry (derived every render) ------------------ */
   133→  const geo: Geo | null = useMemo(
   134→    () =>
   135→      layout ? computeGeo(layout, t.scale, commits.length) : null,
   136→    [layout, t.scale, commits.length],
   137→  )
   138→
   139→  const isNarrow = size.w < 640
   140→  const laneAreaW = geo
   141→    ? Math.min(
   142→        Math.max(140, Math.min(isNarrow ? size.w * 0.5 : size.w * 0.55, 460)),
   143→        Math.max(geo.contentW, 90),
   144→      )
   145→    : 140
   146→
   147→  const viewH = size.h
   148→  const effTx = geo
   149→    ? geo.contentW <= laneAreaW
   150→      ? (laneAreaW - geo.contentW) / 2
   151→      : clamp(t.tx, laneAreaW - geo.contentW, 0)
   152→    : 0
   153→  // screen y = world y - ty  →  ty ∈ [0, totalH - viewH] when content is
   154→  // taller than the viewport; when shorter, content is centered vertically.
   155→  const effTy = geo
   156→    ? geo.totalH <= viewH || viewH === 0
   157→      ? (geo.totalH - viewH) / 2
   158→      : clamp(t.ty, 0, geo.totalH - viewH)
   159→    : 0
   160→
   161→  const txRef = useRef(effTx)
   162→  const tyRef = useRef(effTy)
   163→
   164→  // Mirror the latest render values into refs for event handlers and
   165→  // async callbacks. This effect flushes after every commit — before any
   166→  // user event can possibly fire — so handlers always see fresh values.
   167→  useEffect(() => {
   168→    tRef.current = t
   169→    sizeRef.current = size
   170→    commitsRef.current = commits
   171→    layoutRef.current = layout
   172→    txRef.current = effTx
   173→    tyRef.current = effTy
   174→  })
   175→
   176→  /* ---------------- resize observer --------------------------------- */
   177→  useEffect(() => {
   178→    const el = containerRef.current
   179→    if (!el) return
   180→    const ro = new ResizeObserver((entries) => {
   181→      const rect = entries[0].contentRect
   182→      if (rect.width > 0) setSize({ w: rect.width, h: rect.height })
   183→    })
   184→    ro.observe(el)
   185→    return () => ro.disconnect()
   186→  }, [])
   187→
   188→  /* ---------------- clamped pan/zoom helpers ------------------------ */
   189→
   190→  const setTransform = useCallback(
   191→    (next: Partial<Transform>) => {
   192→      setT((prev) => ({ ...prev, ...next }))
   193→    },
   194→    [],
   195→  )
   196→
   197→  const zoomAt = useCallback(
   198→    (px: number, py: number, factor: number) => {
   199→      const prev = tRef.current
   200→      const newScale = clamp(prev.scale * factor, MIN_SCALE, MAX_SCALE)
   201→      const k = newScale / prev.scale
   202→      if (k === 1) return
   203→      setT({
   204→        scale: newScale,
   205→        tx: px - (px - prev.tx) * k,
   206→        ty: (py + prev.ty) * k - py,
   207→      })
   208→    },
   209→    [],
   210→  )
   211→
   212→  const animateTo = useCallback(
   213→    (target: Pick<Transform, 'ty'> & Partial<Pick<Transform, 'tx'>>) => {
   214→      cancelPanAnim()
   215→      const start = { ...tRef.current }
   216→      const t0 = performance.now()
   217→      const dur = 280
   218→      const step = (now: number) => {
   219→        const p = Math.min(1, (now - t0) / dur)
   220→        const e = 1 - Math.pow(1 - p, 3)
   221→        // merge partial transform — never drop scale via raw setT
   222→        setTransform({
   223→          ty: start.ty + (target.ty - start.ty) * e,
   224→          ...(target.tx !== undefined
   225→            ? { tx: start.tx + (target.tx - start.tx) * e }
   226→            : {}),
   227→        })
   228→        panAnim.current = p < 1 ? requestAnimationFrame(step) : null
   229→      }
   230→      panAnim.current = requestAnimationFrame(step)
   231→    },
   232→    [cancelPanAnim, setTransform],
   233→  )
   234→
   235→  const computeFitScale = useCallback((): number => {
   236→    const lay = layoutRef.current
   237→    const { w } = sizeRef.current
   238→    if (!lay) return 1
   239→    const narrow = w < 640
   240→    const avail = Math.max(140, Math.min(narrow ? w * 0.5 : w * 0.55, 460))
   241→    const laneWidths = lay.laneCount * 34
   242→    const scale = (avail - 18 - 14) / Math.max(laneWidths, 1)
   243→    return clamp(scale, MIN_SCALE, Math.min(MAX_SCALE, 1.35))
   244→  }, [])
   245→
   246→  const fit = useCallback(() => {
   247→    const fitScale = computeFitScale()
   248→    const k = fitScale / tRef.current.scale
   249→    if (k !== 1) {
   250→      zoomAt(laneAreaW / 2, sizeRef.current.h / 2, k)
   251→    }
   252→    // re-center horizontally
   253→    const lay = layoutRef.current
   254→    if (lay) {
   255→      const g = computeGeo(lay, fitScale, commitsRef.current.length)
   256→      const areaW = laneAreaW
   257→      const targetTx =
   258→        g.contentW <= areaW ? (areaW - g.contentW) / 2 : clamp(0, areaW - g.contentW, 0)
   259→      setTransform({ tx: targetTx })
   260→    }
   261→  }, [zoomAt, setTransform, computeFitScale, laneAreaW])
   262→
   263→  /* ---------------- initial fit + intro on new data ------------------ */
   264→  const sizeReady = size.w > 40
   265→
   266→  useEffect(() => {
   267→    if (!layout || !sizeReady) return
   268→    const raf = requestAnimationFrame(() => {
   269→      const fitScale = computeFitScale()
   270→      setT({ scale: fitScale, tx: 0, ty: 0 })
   271→    })
   272→    const timer = setTimeout(() => setIntro(true), 30)
   273→    const timer2 = setTimeout(() => setIntro(false), 1700)
   274→    return () => {
   275→      cancelAnimationFrame(raf)
   276→      clearTimeout(timer)
   277→      clearTimeout(timer2)
   278→    }
   279→  }, [layout, sizeReady, computeFitScale])
   280→
   281→  /* ---------------- center on externally selected commit ------------- */
   282→  useEffect(() => {
   283→    if (!layout || !selectedHash) return
   284→    const row = layout.rowOf.get(selectedHash)
   285→    if (row === undefined) return
   286→    const raf = requestAnimationFrame(() => {
   287→      const lay = layoutRef.current
   288→      const tr = tRef.current
   289→      if (!lay) return
   290→      const g = computeGeo(lay, tr.scale, commitsRef.current.length)
   291→      const h = sizeRef.current.h
   292→      const y = 14 + row * g.rh + g.rh / 2 - tr.ty
   293→      const lane = lay.laneOf.get(selectedHash) ?? 0
   294→      const x = laneX(lane, g) + txRef.current
   295→
   296→      const next: { ty?: number; tx?: number } = {}
   297→      if (y < 80 || y > h - 80) {
   298→        next.ty = 14 + row * g.rh + g.rh / 2 - h / 2
   299→      }
   300→      const areaW = Math.min(
   301→        Math.max(140, Math.min(sizeRef.current.w < 640 ? sizeRef.current.w * 0.5 : sizeRef.current.w * 0.55, 460)),
   302→        Math.max(g.contentW, 90),
   303→      )
   304→      if (x < 48) next.tx = txRef.current + (48 - x)
   305→      else if (x > areaW - 48) next.tx = txRef.current - (x - (areaW - 48))
   306→      if (next.ty !== undefined) {
   307→        animateTo({ ty: next.ty, ...(next.tx !== undefined ? { tx: next.tx } : {}) })
   308→      } else if (next.tx !== undefined) {
   309→        animateTo({ ty: tr.ty, tx: next.tx })
   310→      }
   311→    })
   312→    return () => cancelAnimationFrame(raf)
   313→  }, [selectedHash, layout, animateTo])
   314→
   315→  /* ---------------- wheel: pan vertically, ctrl = zoom --------------- */
   316→  useEffect(() => {
   317→    const el = containerRef.current
   318→    if (!el) return
   319→    const onWheel = (e: WheelEvent) => {
   320→      e.preventDefault()
   321→      cancelPanAnim()
   322→      const rect = el.getBoundingClientRect()
   323→      const px = e.clientX - rect.left
   324→      const py = e.clientY - rect.top
   325→      if (e.ctrlKey || e.metaKey) {
   326→        zoomAt(px, py, Math.exp(-e.deltaY * 0.0022))
   327→      } else {
   328→        setT((prev) => ({ ...prev, ty: prev.ty + e.deltaY, tx: prev.tx + e.deltaX }))
   329→      }
   330→    }
   331→    el.addEventListener('wheel', onWheel, { passive: false })
   332→    return () => el.removeEventListener('wheel', onWheel)
   333→  }, [zoomAt, cancelPanAnim])
   334→
   335→  /* ---------------- pointer pan / pinch ------------------------------ */
   336→  /* window-level move/up listeners are installed once; they no-op unless
   337→     an interaction is active (the pointers map is populated by the
   338→     container's pointerdown). This avoids stale closures entirely. */
   339→
   340→  useEffect(() => {
   341→    const onMove = (e: PointerEvent) => {
   342→      const pts = pointers.current
   343→      if (!pts.has(e.pointerId)) return
   344→      pts.set(e.pointerId, { x: e.clientX, y: e.clientY })
   345→
   346→      if (pts.size === 1 && panStart.current) {
   347→        const s = panStart.current
   348→        const dx = e.clientX - s.x
   349→        const dy = e.clientY - s.y
   350→        if (Math.abs(dx) + Math.abs(dy) > 4) didPan.current = true
   351→        if (moveScheduled.current) return
   352→        moveScheduled.current = true
   353→        const fdx = dx
   354→        const fdy = dy
   355→        requestAnimationFrame(() => {
   356→          moveScheduled.current = false
   357→          // dragging content: finger down (fdy > 0) moves content down,
   358→          // i.e. screen y grows → ty shrinks
   359→          setT((prev) => ({ ...prev, tx: s.tx + fdx, ty: s.ty - fdy }))
   360→        })
   361→      } else if (pts.size === 2 && pinchStart.current) {
   362→        const ps = pinchStart.current
   363→        const [a, b] = [...pts.values()]
   364→        const dist = Math.hypot(a.x - b.x, a.y - b.y)
   365→        const midX = (a.x + b.x) / 2 - (containerRef.current?.getBoundingClientRect().left ?? 0)
   366→        const midY = (a.y + b.y) / 2 - (containerRef.current?.getBoundingClientRect().top ?? 0)
   367→        const newScale = clamp(ps.scale * (dist / ps.dist), MIN_SCALE, MAX_SCALE)
   368→        const k = newScale / ps.scale
   369→        didPan.current = true
   370→        setT({
   371→          scale: newScale,
   372→          tx: midX - (ps.midX - ps.tx) * k,
   373→          ty: (ps.midY + ps.ty) * k - midY,
   374→        })
   375→      }
   376→    }
   377→
   378→    const onUp = (e: PointerEvent) => {
   379→      pointers.current.delete(e.pointerId)
   380→      if (pointers.current.size < 2) pinchStart.current = null
   381→      if (pointers.current.size === 0) panStart.current = null
   382→    }
   383→
   384→    window.addEventListener('pointermove', onMove)
   385→    window.addEventListener('pointerup', onUp)
   386→    window.addEventListener('pointercancel', onUp)
   387→    return () => {
   388→      window.removeEventListener('pointermove', onMove)
   389→      window.removeEventListener('pointerup', onUp)
   390→      window.removeEventListener('pointercancel', onUp)
   391→    }
   392→  }, [])
   393→
   394→  const onPointerDown = useCallback(
   395→    (e: React.PointerEvent) => {
   396→      if (e.button !== 0) return
   397→      cancelPanAnim()
   398→      const rect = containerRef.current?.getBoundingClientRect()
   399→      if (!rect) return
   400→      pointers.current.set(e.pointerId, { x: e.clientX, y: e.clientY })
   401→      if (pointers.current.size === 1) {
   402→        panStart.current = {
   403→          tx: txRef.current,
   404→          ty: tyRef.current,
   405→          x: e.clientX,
   406→          y: e.clientY,
   407→        }
   408→        didPan.current = false
   409→      } else if (pointers.current.size === 2) {
   410→        const [a, b] = [...pointers.current.values()]
   411→        pinchStart.current = {
   412→          dist: Math.hypot(a.x - b.x, a.y - b.y),
   413→          scale: tRef.current.scale,
   414→          midX: (a.x + b.x) / 2 - rect.left,
   415→          midY: (a.y + b.y) / 2 - rect.top,
   416→          tx: tRef.current.tx,
   417→          ty: tRef.current.ty,
   418→        }
   419→        panStart.current = null
   420→      }
   421→    },
   422→    [cancelPanAnim],
   423→  )
   424→
   425→  const onDoubleClick = useCallback(
   426→    (e: React.MouseEvent) => {
   427→      const rect = containerRef.current?.getBoundingClientRect()
   428→      if (!rect) return
   429→      zoomAt(e.clientX - rect.left, e.clientY - rect.top, 1.35)
   430→    },
   431→    [zoomAt],
   432→  )
   433→
   434→  /* ---------------- hover / tooltip ---------------------------------- */
   435→
   436→  const rowFromClientY = useCallback((clientY: number): number | null => {
   437→    const lay = layoutRef.current
   438→    if (!lay) return null
   439→    const geoNow = computeGeo(lay, tRef.current.scale, commitsRef.current.length)
   440→    const y = clientY + tyRef.current - 14 // world y
   441→    const r = Math.floor((y - geoNow.rh / 2) / geoNow.rh)
   442→    return r >= 0 && r < commitsRef.current.length ? r : null
   443→  }, [])
   444→
   445→  const onMouseMove = useCallback(
   446→    (e: React.MouseEvent) => {
   447→      if (pointers.current.size > 0) return // dragging
   448→      const row = rowFromClientY(e.clientY)
   449→      setHoveredRow((prev) => (prev === row ? prev : row))
   450→      // position tooltip via DOM (no re-render)
   451→      const el = containerRef.current
   452→      const tip = tooltipRef.current
   453→      if (el && tip) {
   454→        const rect = el.getBoundingClientRect()
   455→        const x = clamp(e.clientX - rect.left + 14, 8, Math.max(8, rect.width - 240))
   456→        const y = clamp(e.clientY - rect.top + 14, 8, Math.max(8, rect.height - 120))
   457→        tip.style.left = `${x}px`
   458→        tip.style.top = `${y}px`
   459→        tip.style.opacity = row !== null ? '1' : '0'
   460→      }
   461→    },
   462→    [rowFromClientY],
   463→  )
   464→
   465→  const onMouseLeave = useCallback(() => {
   466→    setHoveredRow(null)
   467→    if (tooltipRef.current) tooltipRef.current.style.opacity = '0'
   468→  }, [])
   469→
   470→  const selectCommit = useCallback(
   471→    (hash: string) => {
   472→      if (!didPan.current) onSelect(hash)
   473→    },
   474→    [onSelect],
   475→  )
   476→
   477→  /* ---------------- visible slice ------------------------------------ */
   478→
   479→  const visible = useMemo(() => {
   480→    if (!layout || !geo || viewH === 0) return null
   481→    const n = layout.nodes.length
   482→    const startRow = Math.max(0, Math.floor((effTy - 14) / geo.rh - 0.5) - VISIBLE_BUFFER)
   483→    const endRow = Math.min(n - 1, Math.ceil((viewH + effTy - 14) / geo.rh - 0.5) + VISIBLE_BUFFER)
   484→    if (startRow > endRow) return null
   485→    const nodes = layout.nodes.slice(startRow, endRow + 1)
   486→    const edges = layout.edges.filter(
   487→      (e) => e.childRow <= endRow + 1 && e.parentRow >= startRow - 1,
   488→    )
   489→    return { startRow, endRow, nodes, edges }
   490→  }, [layout, geo, viewH, effTy])
   491→
   492→  const hoveredCommit =
   493→    hoveredRow !== null && layout ? layout.nodes[hoveredRow]?.commit : undefined
   494→
   495→  const gutterMarks = useMemo(() => {
   496→    if (!layout || !geo || !visible) return []
   497→    const out: Array<{ y: number; label: string; strong: boolean }> = []
   498→    let lastY = -Infinity
   499→    const showMonths = geo.rh >= 19
   500→    for (const m of layout.monthMarks) {
   501→      if (m.row < visible.startRow - 2 || m.row > visible.endRow + 2) continue
   502→      if (!m.isYearStart && !showMonths) continue
   503→      const y = rowY(m.row, geo) - effTy
   504→      if (y - lastY < 16) continue
   505→      lastY = y
   506→      out.push({
   507→        y,
   508→        label: m.isYearStart ? String(m.year) : MONTHS[m.month],
   509→        strong: m.isYearStart,
   510→      })
   511→    }
   512→    return out
   513→  }, [layout, geo, visible, effTy])
   514→
   515→  const zoomPct = Math.round(t.scale * 100)
   516→
   517→  /* ---------------- render ------------------------------------------- */
   518→
   519→  return (
   520→    <div className="flex min-w-0 flex-col">
   521→      {/* toolbar */}
   522→      <div className="flex flex-wrap items-center gap-2 border-b px-3 py-2">
   523→        {extra}
   524→        <div className="ml-auto flex items-center gap-1">
   525→          {fetching && (
   526→            <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin text-muted-foreground" />
   527→          )}
   528→          <Button
   529→            variant="outline"
   530→            size="icon"
   531→            className="h-7 w-7"
   532→            aria-label="Zoom out"
   533→            onClick={() =>
   534→              zoomAt(laneAreaW / 2, size.h / 2, 1 / 1.25)
   535→            }
   536→          >
   537→            <ZoomOut className="h-3.5 w-3.5" />
   538→          </Button>
   539→          <span className="w-10 text-center text-[11px] font-medium tabular-nums text-muted-foreground">
   540→            {zoomPct}%
   541→          </span>
   542→          <Button
   543→            variant="outline"
   544→            size="icon"
   545→            className="h-7 w-7"
   546→            aria-label="Zoom in"
   547→            onClick={() => zoomAt(laneAreaW / 2, size.h / 2, 1.25)}
   548→          >
   549→            <ZoomIn className="h-3.5 w-3.5" />
   550→          </Button>
   551→          <Button
   552→            variant="outline"
   553→            size="icon"
   554→            className="h-7 w-7"
   555→            aria-label="Fit to screen"
   556→            onClick={fit}
   557→          >
   558→            <Maximize2 className="h-3.5 w-3.5" />
   559→          </Button>
   560→          <Button
   561→            variant="outline"
   562→            size="icon"
   563→            className="h-7 w-7"
   564→            aria-label="Jump to latest commits"
